\documentclass[tikz,border=8pt]{standalone}
\usetikzlibrary{arrows.meta,positioning,shapes.misc}

\input{grid_selection.texinc}

\definecolor{gridpoint}{RGB}{150,150,150}
\definecolor{spaceblue}{RGB}{32,105,180}
\definecolor{slopegreen}{RGB}{32,145,105}

\begin{document}
\begin{tikzpicture}[
    x=1.50cm,
    y=1.50cm,
    scale=1.5,
    transform shape,
    >=Latex,
    node/.style={circle,draw=gridpoint,fill=gridpoint,inner sep=2.4pt},
    midpoint/.style={cross out,draw,line width=1.2pt,minimum size=7pt},
    every node/.style={font=\small}
]
  \pgfmathtruncatemacro{\previousi}{\selectedi-1}
  \pgfmathtruncatemacro{\nexti}{\selectedi+1}
  \pgfmathsetmacro{\lefthalf}{\selectedi-0.5}
  \pgfmathsetmacro{\righthalf}{\selectedi+0.5}
  \pgfmathsetmacro{\slopelabeln}{\selectedn+0.72}
  \pgfmathsetmacro{\derivativelabeln}{\selectedn+1.48}

  \draw[->,line width=0.7pt] (-0.35,0) -- (4.45,0) node[right] {position $x$};
  \draw[->,line width=0.7pt] (0,-0.35) -- (0,3.45) node[above] {time $t$};

  \foreach \gridi in {0,...,4}
    \foreach \gridn in {0,...,3}
      \node[node] at (\gridi,\gridn) {};

  \draw[spaceblue,line width=2.6pt,line cap=round]
    (\previousi,\selectedn) -- (\selectedi,\selectedn);
  \draw[spaceblue,line width=2.6pt,line cap=round]
    (\selectedi,\selectedn) -- (\nexti,\selectedn);
  \node[midpoint,spaceblue] (leftslope) at (\lefthalf,\selectedn) {};
  \node[midpoint,spaceblue] (rightslope) at (\righthalf,\selectedn) {};
  \node[below=5pt] at (\previousi,\selectedn) {$\theta_{i-1}^n$};
  \node[below=5pt] at (\selectedi,\selectedn) {$\theta_i^n$};
  \node[below=5pt] at (\nexti,\selectedn) {$\theta_{i+1}^n$};
  \node[spaceblue,above=7pt] at (\lefthalf,\selectedn) {$q_{i-1/2}^n$};
  \node[spaceblue,above=7pt] at (\righthalf,\selectedn) {$q_{i+1/2}^n$};

  \node[draw=slopegreen,circle,line width=1.5pt,minimum size=8pt]
    (center) at (\selectedi,\selectedn) {};
  \draw[slopegreen,->,line width=1.2pt,bend left=32]
    (leftslope.north) to (center.north west);
  \draw[slopegreen,->,line width=1.2pt,bend right=32]
    (rightslope.north) to (center.north east);
  \node[slopegreen] at (\selectedi,\slopelabeln)
    {difference of neighboring slopes};
  \node[slopegreen,align=center,text width=5.4cm]
    at (\selectedi,\derivativelabeln)
    {$\left.\partial^2\theta/\partial x^2\right|_i^n$\\returns to the stored grid point $i$};
\end{tikzpicture}
\end{document}
